## 7. トリガー関数

### ユーザー作成時の自動同期
```sql
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER AS $$
BEGIN
  -- public.usersにレコードを作成
  INSERT INTO public.users (id, email, username)
  VALUES (
    NEW.id,
    NEW.email,
    COALESCE(NEW.raw_user_meta_data->>'username', split_part(NEW.email, '@', 1))
  );
  
  -- user_statsを初期化
  INSERT INTO public.user_stats (user_id)
  VALUES (NEW.id);
  
  -- デフォルトロールを付与
  INSERT INTO public.user_roles (user_id, role)
  VALUES (NEW.id, 'user');
  
  -- 監査ログに記録
  INSERT INTO public.audit_logs (user_id, action, metadata)
  VALUES (NEW.id, 'user_created', jsonb_build_object('source', 'auth_trigger'));
  
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- トリガーの設定
CREATE TRIGGER on_auth_user_created
AFTER INSERT ON auth.users
FOR EACH ROW EXECUTE FUNCTION public.handle_new_user();
```

### 統計情報の自動更新
```sql
CREATE OR REPLACE FUNCTION public.update_user_stats()
RETURNS TRIGGER AS $$
BEGIN
  IF TG_TABLE_NAME = 'posts' THEN
    IF TG_OP = 'INSERT' THEN
      UPDATE public.user_stats 
      SET post_count = post_count + 1,
          updated_at = NOW()
      WHERE user_id = NEW.user_id;
    ELSIF TG_OP = 'DELETE' THEN
      UPDATE public.user_stats 
      SET post_count = post_count - 1,
          updated_at = NOW()
      WHERE user_id = OLD.user_id;
    END IF;
  END IF;
  
  RETURN NULL;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER update_post_count
AFTER INSERT OR DELETE ON public.posts
FOR EACH ROW EXECUTE FUNCTION public.update_user_stats();
```

## 8. Row Level Security (RLS) ポリシー

```sql
-- RLSを有効化
ALTER TABLE public.users ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.posts ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.shrines ENABLE ROW LEVEL SECURITY;

-- ユーザー情報のポリシー
CREATE POLICY "Users can view all profiles" 
  ON public.users FOR SELECT 
  USING (true);

CREATE POLICY "Users can update own profile" 
  ON public.users FOR UPDATE 
  USING (auth.uid() = id);

-- 投稿のポリシー
CREATE POLICY "Public posts are viewable by everyone" 
  ON public.posts FOR SELECT 
  USING (is_published = true);

CREATE POLICY "Users can manage own posts" 
  ON public.posts FOR ALL 
  USING (auth.uid() = user_id);

-- 神社情報のポリシー
CREATE POLICY "Approved shrines are viewable by everyone" 
  ON public.shrines FOR SELECT 
  USING (status = 'approved');

CREATE POLICY "Users can create shrine submissions" 
  ON public.shrines FOR INSERT 
  WITH CHECK (auth.uid() = created_by_user_id);

-- 管理者は全ての神社を管理可能
CREATE POLICY "Admins can manage all shrines" 
  ON public.shrines FOR ALL 
  USING (
    EXISTS (
      SELECT 1 FROM public.user_roles 
      WHERE user_id = auth.uid() 
      AND role IN ('admin', 'super_admin', 'moderator')
    )
  );
```

## 9. マイグレーション手順

```sql
-- 1. 既存データのバックアップ
-- 2. 新しいスキーマの作成
-- 3. データの移行（既存のBIGINT IDをUUIDにマッピング）

-- 移行用の一時テーブル
CREATE TABLE migration_id_mapping (
  old_id BIGINT,
  new_id UUID,
  table_name TEXT,
  PRIMARY KEY (table_name, old_id)
);

-- 移行スクリプトの例
INSERT INTO migration_id_mapping (old_id, new_id, table_name)
SELECT id, gen_random_uuid(), 'users' FROM old_users;

-- 新しいテーブルへのデータ挿入
INSERT INTO public.users (id, username, email, profile_image_url, bio, is_active, created_at, updated_at)
SELECT 
  m.new_id,
  u.username,
  u.email,
  u.profile_image_url,
  u.bio,
  u.is_active,
  u.created_at,
  u.updated_at
FROM old_users u
JOIN migration_id_mapping m ON m.old_id = u.id AND m.table_name = 'users';
```

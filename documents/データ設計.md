## 12. トリガー関数の作成

### ユーザー作成時の自動同期
```sql
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER AS $$
BEGIN
  -- public.usersにレコードを作成
  INSERT INTO public.users (id, email, username)
  VALUES (
    NEW.id,
    NEW.email,
    COALESCE(NEW.raw_user_meta_data->>'username', split_part(NEW.email, '@', 1))
  );
  
  -- user_statsを初期化
  INSERT INTO public.user_stats (user_id)
  VALUES (NEW.id);
  
  -- デフォルトロールを付与
  INSERT INTO public.user_roles (user_id, role)
  VALUES (NEW.id, 'user');
  
  -- 監査ログに記録
  INSERT INTO public.audit_logs (user_id, action, metadata)
  VALUES (NEW.id, 'user_created', jsonb_build_object('source', 'auth_trigger'));
  
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- トリガーの設定
CREATE TRIGGER on_auth_user_created
AFTER INSERT ON auth.users
FOR EACH ROW EXECUTE FUNCTION public.handle_new_user();
```

### 統計情報の自動更新
```sql
CREATE OR REPLACE FUNCTION public.update_user_stats_on_post()
RETURNS TRIGGER AS $$
BEGIN
  IF TG_OP = 'INSERT' THEN
    -- 投稿数を増やす
    UPDATE public.user_stats 
    SET post_count = post_count + 1
    WHERE user_id = NEW.user_id;
  ELSIF TG_OP = 'DELETE' THEN
    -- 投稿数を減らす
    UPDATE public.user_stats 
    SET post_count = GREATEST(post_count - 1, 0)
    WHERE user_id = OLD.user_id;
  END IF;
  
  RETURN NULL;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER update_post_count
AFTER INSERT OR DELETE ON public.posts
FOR EACH ROW EXECUTE FUNCTION public.update_user_stats_on_post();

-- いいね数の更新
CREATE OR REPLACE FUNCTION public.update_likes_count()
RETURNS TRIGGER AS $$
BEGIN
  IF TG_OP = 'INSERT' THEN
    UPDATE public.user_stats us
    SET total_likes_received = total_likes_received + 1
    FROM public.posts p
    WHERE p.id = NEW.post_id AND p.user_id = us.user_id;
  ELSIF TG_OP = 'DELETE' THEN
    UPDATE public.user_stats us
    SET total_likes_received = GREATEST(total_likes_received - 1, 0)
    FROM public.posts p
    WHERE p.id = OLD.post_id AND p.user_id = us.user_id;
  END IF;
  
  RETURN NULL;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER update_likes_count
AFTER INSERT OR DELETE ON public.post_likes
FOR EACH ROW EXECUTE FUNCTION public.update_likes_count();

-- フォロー数の更新
CREATE OR REPLACE FUNCTION public.update_follow_counts()
RETURNS TRIGGER AS $$
BEGIN
  IF TG_OP = 'INSERT' THEN
    -- フォロワー数を増やす
    UPDATE public.user_stats 
    SET follower_count = follower_count + 1
    WHERE user_id = NEW.following_user_id;
    
    -- フォロー数を増やす
    UPDATE public.user_stats 
    SET following_count = following_count + 1
    WHERE user_id = NEW.follower_user_id;
  ELSIF TG_OP = 'DELETE' THEN
    -- フォロワー数を減らす
    UPDATE public.user_stats 
    SET follower_count = GREATEST(follower_count - 1, 0)
    WHERE user_id = OLD.following_user_id;
    
    -- フォロー数を減らす
    UPDATE public.user_stats 
    SET following_count = GREATEST(following_count - 1, 0)
    WHERE user_id = OLD.follower_user_id;
  END IF;
  
  RETURN NULL;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER update_follow_counts
AFTER INSERT OR DELETE ON public.follows
FOR EACH ROW EXECUTE FUNCTION public.update_follow_counts();
```

## 13. Row Level Security (RLS) ポリシー

```sql
-- RLSを有効化
ALTER TABLE public.users ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.posts ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.shrines ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.goshuin_types ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.post_likes ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.follows ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.user_badges ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.subscriptions ENABLE ROW LEVEL SECURITY;

-- ユーザー情報のポリシー
CREATE POLICY "Users can view all profiles" 
  ON public.users FOR SELECT 
  USING (true);

CREATE POLICY "Users can update own profile" 
  ON public.users FOR UPDATE 
  USING (auth.uid() = id);

-- 投稿のポリシー
CREATE POLICY "Public posts are viewable by everyone" 
  ON public.posts FOR SELECT 
  USING (is_published = true);

CREATE POLICY "Users can manage own posts" 
  ON public.posts FOR ALL 
  USING (auth.uid() = user_id);

-- 神社情報のポリシー
CREATE POLICY "Approved shrines are viewable by everyone" 
  ON public.shrines FOR SELECT 
  USING (status = 'approved');

CREATE POLICY "Users can create shrine submissions" 
  ON public.shrines FOR INSERT 
  WITH CHECK (auth.uid() = created_by_user_id);

-- 御朱印情報のポリシー  
CREATE POLICY "Approved goshuin types are viewable by everyone" 
  ON public.goshuin_types FOR SELECT 
  USING (status = 'approved');

CREATE POLICY "Users can create goshuin submissions" 
  ON public.goshuin_types FOR INSERT 
  WITH CHECK (auth.uid() = created_by_user_id);

-- いいねのポリシー
CREATE POLICY "Likes are viewable by everyone" 
  ON public.post_likes FOR SELECT 
  USING (true);

CREATE POLICY "Users can manage own likes" 
  ON public.post_likes FOR INSERT OR DELETE 
  USING (auth.uid() = user_id);

-- フォローのポリシー
CREATE POLICY "Follows are viewable by everyone" 
  ON public.follows FOR SELECT 
  USING (true);

CREATE POLICY "Users can manage own follows" 
  ON public.follows FOR INSERT OR DELETE 
  USING (auth.uid() = follower_user_id);

-- バッジのポリシー
CREATE POLICY "Badges are viewable by everyone" 
  ON public.user_badges FOR SELECT 
  USING (true);

-- サブスクリプションのポリシー
CREATE POLICY "Users can view own subscription" 
  ON public.subscriptions FOR SELECT 
  USING (auth.uid() = user_id);

-- 管理者ポリシー
CREATE POLICY "Admins can manage all shrines" 
  ON public.shrines FOR ALL 
  USING (
    EXISTS (
      SELECT 1 FROM public.user_roles 
      WHERE user_id = auth.uid() 
      AND role IN ('admin', 'super_admin', 'moderator')
    )
  );

CREATE POLICY "Admins can manage all goshuin types" 
  ON public.goshuin_types FOR ALL 
  USING (
    EXISTS (
      SELECT 1 FROM public.user_roles 
      WHERE user_id = auth.uid() 
      AND role IN ('admin', 'super_admin', 'moderator')
    )
  );
```
